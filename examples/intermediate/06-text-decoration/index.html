Text Decoration

text-decoration

text-decoration: none | <line> <style> <color>

none — no decoration (default)

underline — line under text

overline — line above text

line-through — line through text

underline wavy red — shorthand with style and color

underline dotted — shorthand with style

text-decoration-line

text-decoration-line: none | underline | overline | line-through

none — no decoration (default)

underline — line under text

overline — line above text

line-through — line through text

underline overline — multiple lines

text-decoration-style

text-decoration-style: solid | double | dotted | dashed | wavy

solid — solid line (default)

dashed — dashed line

dotted — dotted line

double — double line

wavy — wavy line

text-decoration-color

text-decoration-color: <color>

currentColor — matches text color (default)

red — named color

#a3a3a3 — hex color

text-decoration-thickness

text-decoration-thickness: auto | from-font | <length> | <percentage>

auto — browser-determined (default)

from-font — use the font's thickness

1px — thin line

3px — thick line

text-underline-offset

text-underline-offset: auto | from-font | <length> | <percentage>

auto — default position

from-font — use the font's offset

1px — close to text

4px — small gap

8px — far from text
