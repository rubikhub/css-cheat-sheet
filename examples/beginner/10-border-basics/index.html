Border Basics

border-width

border-width: <length> | thin | medium | thick

thin

thin — thin border

medium

medium — medium border (default)

thick

thick — thick border

1px

1px — pixel width

2px

2px — thicker border

0

0 — no border

Global keywords also apply: inherit · initial (medium) · revert · unset.

border-style

border-style: none | solid | dashed | dotted | double | groove | ridge | inset | outset

none

none — no border (default)

solid

solid — solid line

dashed

dashed — dashed line

dotted

dotted — dotted line

double

double — double line

groove

groove — 3D grooved effect

ridge

ridge — 3D ridged effect

inset

inset — 3D inset effect

outset

outset — 3D outset effect

Global keywords also apply: inherit · initial (none) · revert · unset.

border-color

border-color: <color>

currentColor

currentColor — matches text color

#ccc

#ccc — hex color

rgb(0, 0, 0)

rgb(0, 0, 0) — RGB color

red

red — named color

transparent

transparent — invisible border

Global keywords also apply: inherit · initial (currentColor) · revert · unset.

border

border: <width> <style> <color>

1px solid #ccc

1px solid #ccc — width style color

2px dashed red

2px dashed red — thick dashed border

none

none — no border (default)

Global keywords also apply: inherit · initial (none medium currentColor) · revert · unset.

border-radius

border-radius: <length> | <percentage>

4px

4px — slight rounding

8px

8px — moderate rounding

50%

50% — fully round (circle/ellipse)

1rem

1rem — rem-based rounding

4px 8px

4px 8px — tl/br | tr/bl

4px 8px 12px 16px

4px 8px 12px 16px — tl | tr | br | bl

0

0 — no rounding (default)

Global keywords also apply: inherit · initial (0) · revert · unset.
